from pathlib import Path
from PIL import Image, ImageEnhance
out = Path("exports")
out.mkdir(exist_ok=True)
for photo in Path("photos").glob("*.jpg"):
    im = Image.open(photo)
    im = ImageEnhance.Contrast(im).enhance(1.15)
    im.save(out / photo.name)
